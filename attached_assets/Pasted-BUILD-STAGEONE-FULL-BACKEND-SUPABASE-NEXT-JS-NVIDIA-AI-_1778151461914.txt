3. DATABASE SCHEMA (SUPABASE SQL)
create table artists (
  id uuid primary key default uuid_generate_v4(),
  email text unique,
  artist_name text,
  genre text,
  secondary_genre text,
  style_tags jsonb default '[]',
  audience_profile text,
  preferences jsonb default '{}',
  created_at timestamp default now()
);

create table artist_dna (
  id uuid primary key default uuid_generate_v4(),
  artist_id uuid references artists(id) on delete cascade,
  tone text,
  visual_style text,
  hook_style text,
  audience_type text,
  identity_summary text,
  updated_at timestamp default now()
);

create table rollouts (
  id uuid primary key default uuid_generate_v4(),
  artist_id uuid references artists(id) on delete cascade,
  song_title text,
  genre text,
  mood text,
  rollout_timeline jsonb,
  hooks jsonb,
  visual_direction jsonb,
  audience_strategy jsonb,
  launch_score jsonb,
  strategic_insight text,
  created_at timestamp default now()
);

create table campaign_shares (
  id uuid primary key default uuid_generate_v4(),
  rollout_id uuid references rollouts(id) on delete cascade,
  platform text,
  created_at timestamp default now()
);
4. NVIDIA AI ENGINE (FIXED VERSION)
/lib/ai/stageone-engine.ts
export async function generateRollout(input: any, artist: any, dna: any) {

  const prompt = `
You are STAGEONE, a senior music marketing creative director.

You build rollout strategies for artists.

Be:
- strategic
- opinionated
- culturally aware
- genre-specific
- non-generic

ARTIST:
${JSON.stringify(artist)}

DNA:
${JSON.stringify(dna)}

INPUT:
${JSON.stringify(input)}

Return ONLY valid JSON:

{
  "rolloutTimeline": [
    {"phase": "TEASE", "strategy": ""},
    {"phase": "BUILD", "strategy": ""},
    {"phase": "RELEASE", "strategy": ""},
    {"phase": "SUSTAIN", "strategy": ""}
  ],
  "hooks": [
    {"type": "curiosity", "text": ""},
    {"type": "emotional", "text": ""},
    {"type": "identity", "text": ""}
  ],
  "visualDirection": {
    "lighting": "",
    "aesthetic": "",
    "scene": "",
    "colorPalette": ""
  },
  "audienceStrategy": [
    {"segment": "", "behavior": ""}
  ],
  "launchScore": {
    "hookVirality": 0,
    "emotionalDepth": 0,
    "replayValue": 0,
    "trendFit": 0,
    "audienceFit": 0
  },
  "strategicInsight": ""
}
`

  const response = await fetch(
    "https://integrate.api.nvidia.com/v1/chat/completions",
    {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${process.env.NVIDIA_API_KEY}`,
        "Content-Type": "application/json"
      },
      body: JSON.stringify({
        model: "meta/llama-3.1-70b-instruct",
        messages: [
          { role: "user", content: prompt }
        ],
        temperature: 0.7,
        max_tokens: 2000
      })
    }
  )

  const data = await response.json()

  const content = data.choices?.[0]?.message?.content

  try {
    return JSON.parse(content)
  } catch (e) {
    return {
      error: "Invalid AI response",
      raw: content
    }
  }
}
5. API ROUTES
/app/api/generate-rollout/route.ts
import { supabaseServer } from "@/lib/supabase/server"
import { generateRollout } from "@/lib/ai/stageone-engine"

export async function POST(req: Request) {
  const body = await req.json()

  const { artistId, songTitle, genre, mood } = body

  const { data: artist } = await supabaseServer
    .from("artists")
    .select("*")
    .eq("id", artistId)
    .single()

  const { data: dna } = await supabaseServer
    .from("artist_dna")
    .select("*")
    .eq("artist_id", artistId)
    .single()

  const rollout = await generateRollout(
    { songTitle, genre, mood },
    artist,
    dna
  )

  return Response.json(rollout)
}
/app/api/save-rollout/route.ts
import { supabaseServer } from "@/lib/supabase/server"

export async function POST(req: Request) {
  const body = await req.json()

  const { artistId, rollout } = body

  const { data } = await supabaseServer
    .from("rollouts")
    .insert({
      artist_id: artistId,
      song_title: rollout.songTitle,
      genre: rollout.genre,
      mood: rollout.mood,
      rollout_timeline: rollout.rolloutTimeline,
      hooks: rollout.hooks,
      visual_direction: rollout.visualDirection,
      audience_strategy: rollout.audienceStrategy,
      launch_score: rollout.launchScore,
      strategic_insight: rollout.strategicInsight
    })

  return Response.json({ success: true, data })
}
